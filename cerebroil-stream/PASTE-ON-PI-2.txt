# Paste ALL of this into the Raspberry Pi Connect terminal (one go) and press Enter.
# It takes about 3 to 8 minutes. At the end it prints a summary: copy everything it prints and send it back.
#
# What it does:
#   A. shows the Pi's network (read only);
#   B. downloads the latest camera service from GitHub into ~/farmos-src and copies it over ~/cerebroil-stream
#      (your .env with the camera login and the token, and the .venv, are NOT touched);
#   C. installs it and starts it as a service that also starts at boot (this uses sudo and installs packages);
#   D. checks that it answers and runs the checker again.
# The cable address (192.168.0.51) is left as it is.

echo; echo "=== A. The Pi's network (read only)"
ip -4 -br addr
ip route | head -6
echo "Wi-Fi name: $(iwgetid -r 2>/dev/null || echo unknown)"
echo "--- can the Pi reach the router and the laptop?"
ping -c 2 -W 1 192.168.1.1 2>&1 | tail -2
ping -c 2 -W 1 192.168.1.43 2>&1 | tail -2

echo; echo "=== B. Getting the latest camera service"
command -v git >/dev/null || sudo apt-get install -y git
rm -rf ~/farmos-src
git clone --quiet --depth 1 --filter=blob:none --sparse -b responsive-login-local-db https://github.com/JushouaS/FarmOs.git ~/farmos-src
git -C ~/farmos-src sparse-checkout set cerebroil-stream
cp -a ~/farmos-src/cerebroil-stream/. ~/cerebroil-stream/
ls ~/cerebroil-stream

echo; echo "=== C. Installing and starting the service"
cd ~/cerebroil-stream && SKIP_ETH=1 bash setup-pi.sh

echo; echo "=== D. Does it answer on the Pi itself?"
sleep 6
curl -s -m 5 http://127.0.0.1:5000/status; echo
bash ~/cerebroil-stream/check-pi.sh
